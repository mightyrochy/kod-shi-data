```python
from pathlib import Path
import collections, csv, hashlib, math, re

data = Path('джерела/дані_корпусу/ISCC-NBS_вікна.txt').read_bytes().replace(b'\r\n', b'\n')
assert hashlib.sha256(data).hexdigest() == (
    'd8b84ef808b6025622dfb7e774c32f732f1cb4a7400f5c65b8a49f1dfc2ba5e5')
rows = list(csv.DictReader([s for s in data.decode().splitlines()
                           if s and not s.startswith('#')], delimiter='\t'))
owner, families = {}, set()
for line in Path('джерела/колір_сім_ї_вікна.md').read_text(encoding='utf-8').splitlines():
    if not re.match(r'^\| `[a-z_]+` \|', line):
        continue
    fields = [s.strip() for s in line.split('|')[1:-1]]
    if len(fields) != 5:
        continue
    family = fields[0].strip('`')
    assert family not in families
    families.add(family)
    numbers = []
    for part in fields[2].split(', '):
        bounds = [int(s) for s in part.split('–')]
        numbers.extend(range(bounds[0], bounds[-1] + 1))
    for number in numbers:
        assert number not in owner
        owner[number] = family
    assert tuple(map(int, fields[3].split('/'))) == (
        len(numbers), sum(int(r['Number']) in numbers for r in rows))
assert len(families) == 12 and set(owner) == set(range(1, 268))
assert len(rows) == 932 and {int(r['Number']) for r in rows} == set(owner)
charts = collections.defaultdict(list)
for row in rows:
    r = {k: float(v) for k, v in row.items() if k not in ('HueInterval', 'Number')}
    assert 1 <= r['Hmin'] < r['Hmax'] <= 101
    assert 0 <= r['Vmin'] < r['Vmax'] <= 10 and 0 <= r['Cmin'] < r['Cmax']
    charts[r['Hmin'], r['Hmax']].append(r)
intervals = sorted(charts)
assert len(charts) == 31 and intervals[0][0] == 1 and intervals[-1][1] == 101
assert all(a[1] == b[0] for a, b in zip(intervals, intervals[1:]))
checked = 0
for rs in charts.values():
    vs = sorted({r[k] for r in rs for k in ('Vmin', 'Vmax')})
    cs = sorted({r[k] for r in rs for k in ('Cmin', 'Cmax') if math.isfinite(r[k])})
    cs.append(cs[-1] + 1)
    assert vs[0] == 0 and vs[-1] == 10 and cs[0] == 0
    for v0, v1 in zip(vs, vs[1:]):
        for c0, c1 in zip(cs, cs[1:]):
            v, c = (v0 + v1) / 2, (c0 + c1) / 2
            assert sum(r['Vmin'] < v < r['Vmax'] and r['Cmin'] < c < r['Cmax']
                       for r in rs) == 1
            checked += 1
assert checked == 2779
print('12 сімей; 267 блоків; 932 клітинки; 31 аркуш; 2779 ділянок; помилок 0')
```
