## Як повторити прицільні проби

Файл доказів поруч містить `проби` (повний текст кожного ізольованого входу, фактичні місця й пункти) та `вибірка` (зерно, 20 рядків і 20 файлів). Приклад відтворення всіх ізольованих входів: передати шлях до дерева перевіреної голови й шлях до JSON-доказів; код імпортує тільки визначення незміненого приладу, JS розбирає його справжнім помічником, реєстр бере з того ж дерева.

```python
import importlib.util, json, subprocess, sys, tempfile
from pathlib import Path
дерево = Path(sys.argv[1]).resolve()
докази = json.loads(Path(sys.argv[2]).read_text(encoding="utf8"))
шлях = дерево / "джерела/прилади/корінь_річ_запис_речі.py"
опис = importlib.util.spec_from_file_location("прилад", шлях)
прилад = importlib.util.module_from_spec(опис)
опис.loader.exec_module(прилад)
реєстр = прилад.Реєстр(str(дерево))
with tempfile.TemporaryDirectory() as тека:
    for проба in докази["проби"]:
        назва, код = проба["проба"], проба["код"]
        if проба["мова"] == "py":
            модуль = прилад.розібрати_py("джерела/" + назва + ".py", код)
        else:
            файл = Path(тека) / (назва + ".js")
            файл.write_text(код, encoding="utf8")
            запуск = subprocess.run([
                "node", "--expose-internals",
                str(шлях.with_name("корінь_річ_запис_речі_js.js")), str(файл)
            ], capture_output=True, text=True, encoding="utf8", check=True)
            розбір = json.loads(запуск.stdout)[str(файл)]
            модуль = прилад.розібрати_js("джерела/показ/" + назва + ".js", розбір)
        рушій = прилад.Рушій([модуль], реєстр)
        аналіз = прилад.Аналіз(рушій, реєстр)
        аналіз.виконати()
        місця = прилад.обчислити(аналіз, str(дерево)).м
        число = len({(ф, р) for ф, р, п in місця})
        пункти = sorted({п for ф, р, п in місця})
        print(назва, "→", число, "пункти", пункти)
        assert число == проба["місць"] and пункти == проба["пункти"]
```
