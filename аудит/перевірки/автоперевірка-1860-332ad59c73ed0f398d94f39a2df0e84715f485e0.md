$nl = "`n"
$cases = [ordered]@{
  'fenced-example' = "Наряд перевірки.$nl$nl``````text${nl}Роль — ПЕРЕВІРКА (приклад)$nl``````${nl}Кінець."
  'bare-eof'       = "Наряд.${nl}Роль — ПЕРЕВІРКА"

  'own-line-crlf'  = "Наряд.`r${nl}Роль — ПЕРЕВІРКА`r${nl}Далі."
  'absent'         = "Наряд без ролі."
}
if (Get-Command Get-TaskRole -ErrorAction SilentlyContinue) { foreach ($k in $cases.Keys) { "role $k => " + $(if ((Get-TaskRole $cases[$k]) -ceq "ПЕРЕВІРКА") { "REVIEW" } else { "EXECUTOR" }) } }
