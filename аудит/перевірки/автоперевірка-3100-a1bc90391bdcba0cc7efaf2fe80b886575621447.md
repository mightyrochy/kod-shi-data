## Вимір ДО → ПІСЛЯ

Проба викликає справжні `Write-ChainFailure`, `Test-Seen`, `Add-OutboxLine` і
`Read-OutboxFile` з бази та голови. Вісім різних синтетичних шляхів, по дві
відмови `UnauthorizedAccessException` у тому самому кадрі `Get-MarkerState`, одна година UTC.


Текст `suppression3100.ps1` для відтворення; `BaseRoot` має містити базу PR,
`HeadRoot` — перевірену голову:

```powershell
param([string]$BaseRoot,[string]$HeadRoot)
$ErrorActionPreference = 'Stop'
[Console]::OutputEncoding = New-Object Text.UTF8Encoding($false)
foreach ($case in @(@{Name='ДО';Root=$BaseRoot;Expected=8},@{Name='ПІСЛЯ';Root=$HeadRoot;Expected=1})) {
    & {
        $dir = Join-Path $env:TEMP ('suppression3100-' + [guid]::NewGuid().ToString('N'))
        . (Join-Path $case.Root 'аудит/адаптери/gemini/gemini_publish.ps1') -WorkDir "$dir/pub" -OutDir "$dir/out" -OutboxDir "$dir/outbox" -UpdateRoot "$dir/update"
        foreach ($d in @($WorkDir,$OutDir,$OutboxDir,"$WorkDir/seen")) { [void][IO.Directory]::CreateDirectory($d) }
        . ([scriptblock]::Create('function Get-MarkerState([string]$Path) { throw [UnauthorizedAccessException]::new($Path) }'))
        foreach ($i in 1..8) {
            $path = Join-Path $dir ("state-$i.txt")
            foreach ($repeat in 1..2) {
                try { Get-MarkerState $path }
                catch { $_.Exception.Data['StatePath'] = $path; Write-ChainFailure $_ }
            }
        }
        $lines = @(Get-ChildItem -LiteralPath $OutboxDir -File)
        $seen = @(Get-ChildItem -LiteralPath "$WorkDir/seen" -File)
        foreach ($f in $lines) { $result = Read-OutboxFile $f.FullName; if ($result[1] -cne '') { throw 'Вихід не прийнято' } }
        $diagnostic = Join-Path $OutDir 'state-failures.jsonl'
        $records = $(if (Test-Path -LiteralPath $diagnostic) { @([IO.File]::ReadAllLines($diagnostic)).Count } else {0})
        "$($case.Name): файлів 8; відмов 16; рядків $($lines.Count); ключів $($seen.Count); записів діагностики $records; вихід прийнято"
        if ($lines.Count -ne $case.Expected -or $seen.Count -ne $case.Expected) { throw 'Мірило приглушення не пройдено' }
    }
}
```
