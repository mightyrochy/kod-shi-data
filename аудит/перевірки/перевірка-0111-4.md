## Проба пулу Е3 (`е3_пул.py`)

```python
import json, pathlib, sys
_К = pathlib.Path(sys.argv[1]).resolve(); sys.path.insert(0, str(_К))
import bridge as B, feed as Ф
вх = dict(json.load(open(_К / "стенд_вх.json", encoding="utf-8")), каталог=Ф.каталог_на_диску(), варіантів=1, гілка=0)
пул = lambda r, сл: [x["id"] for x in ((r.get("кандидати") or {}).get(сл) or [])]
донор = lambda r, сл: [x["id"] for x in ((r.get("кандидати_донора") or {}).get(сл) or [])]
без = json.loads(B.виклик("запити", json.dumps(вх, ensure_ascii=False)))
слот, ціль = [(с, x) for с in ("сумка", "взуття", "верх", "низ") for x in пул(без, с) if x in донор(без, с)][0]
for н, нц in (("словник р1", {"1": [ціль]}), ("словник р2", {"2": [ціль]}), ("список", [ціль])):
    з = json.loads(B.виклик("запити", json.dumps(dict(вх, не_ця=нц), ensure_ascii=False)))
    print(н, "→ у пулі р1:", ціль in пул(з, слот), "· у пулі р2:", ціль in донор(з, слот))
```
