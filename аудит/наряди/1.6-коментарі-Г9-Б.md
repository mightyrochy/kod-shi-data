**Як.**
1. Перед правкою файлу — `python3 джерела/прилади/лічба_коментарів_історії.py <файл>`: адреси збігів.
2. Знімай із коментаря й докстрінга посилання на рядок дошки, PR, дату, живий прогін, «було», «доти», «тепер», «ЩО ТУТ
   БУЛО»; пояснення «що й чому» лишається; коментар, що без них нічого не пояснює, — геть. Мітка правила лишається,

   лишається, коли пояснює поріг чи вибір; дата й SHA виміру — геть.
3. Код не змінюється. Байти поза коментарями не чіпай: кодування, BOM, LF/CRLF, кінцевий newline, відступи; без

4. Перевірка — скрипт поза репозиторієм (`../g9b.py`, теку вище клону), з кореня клону `python3 ../g9b.py`: кожен файл
   «ГАРАЗД» — нема правок у зонах `ЗАБОРОНЕНО` (рядки `BASE`), `.py` — AST без докстрінгів тотожний `BASE`.
   «ПОМИЛКА» — поверни названі рядки й повтори.
   ```python
   import ast, re, subprocess, sys
   BASE = "96248896e3b89055121fd8b534cbfd79903e63a8"
   # Рядки BASE, яких не чіпати: правки відкритих PR (±3 рядки) і ділянки наступних кроків.
   ЗАБОРОНЕНО = {
       "джерела/outfit.py": "48-61 162-169 193-199 268-281 288-298 305-311 315-321 403-409 427-433 600-606 630-815 "
                            "862-875 894-919 945-958",
       "джерела/суд_річ.py": "18-25 88-94 148-154 184-193 213-230 420-460",
       "джерела/profile.py": "24-38 130-165 175-182 225-233 336-347 410-429 442-448 462-470 520-600",
   }
   def tree(src):
       try:
           t = ast.parse(src)
       except SyntaxError as e:
           return f"синтаксис: {e}"
       for n in ast.walk(t):
           b = getattr(n, "body", None)
           if isinstance(b, list) and b and isinstance(b[0], ast.Expr) \
                   and isinstance(b[0].value, ast.Constant) and isinstance(b[0].value.value, str):
               n.body = b[1:] or [ast.Pass()]
       return ast.dump(t)
   def html(src):
       s = re.sub(r"<!--.*?-->", "", src.decode("utf-8"), flags=re.S)
       return [l.rstrip() for l in s.splitlines() if l.strip()]
   погано = 0
   for f, текст in ЗАБОРОНЕНО.items():
       зони = [tuple(int(v) for v in (x + "-" + x if "-" not in x else x).split("-")) for x in текст.split()]
       d = subprocess.run(["git", "diff", "-U0", BASE, "--", f], capture_output=True, text=True, check=True).stdout
       торкнуто = []
       for h in re.split(r"(?m)^(?=@@ )", d)[1:]:
           m = re.match(r"@@ -(\d+)(?:,(\d+))? ", h)
           a, n = int(m[1]), int(m[2] or 1)
           мінус = [l[1:] for l in h.split("\n")[1:] if l.startswith("-")]
           рядки = [a + i for i, l in enumerate(мінус) if l.strip()] if n else [a]
           торкнуто += [r for r in рядки if any(x <= r <= y for x, y in зони)]
       old = subprocess.run(["git", "show", f"{BASE}:{f}"], capture_output=True, check=True).stdout
       new = open(f, "rb").read()
       той_самий = html(old) == html(new) if f.endswith(".html") else tree(old) == tree(new)
       погано += bool(торкнуто) or not той_самий
       print(f, "ГАРАЗД" if not торкнуто and той_самий else "ПОМИЛКА", "" if той_самий else "код змінено",
             ("заборонені рядки: " + ", ".join(map(str, торкнуто))) if торкнуто else "")
   sys.exit(1 if погано else 0)
   ```
