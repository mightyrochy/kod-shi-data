   ```
   import ast, subprocess, sys
   def tree(src):
       t = ast.parse(src)
       for n in ast.walk(t):
           b = getattr(n, "body", None)
           if isinstance(b, list) and b and isinstance(b[0], ast.Expr) \
                   and isinstance(b[0].value, ast.Constant) and isinstance(b[0].value.value, str):
               n.body = b[1:] or [ast.Pass()]
       return ast.dump(t)
   f = sys.argv[1]
   old = subprocess.run(["git", "show", "origin/main:" + f], capture_output=True, check=True).stdout
   print(f, "AST тотожний" if tree(old) == tree(open(f, "rb").read()) else "AST РІЗНИЙ")
   ```
