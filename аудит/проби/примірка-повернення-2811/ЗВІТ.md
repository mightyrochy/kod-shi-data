## Повтор

Встановити залежність `playwright` і покласти в `$MP` чотири файли з адрес `ФАЙЛИ_СЕГМЕНТАЦІЇ_П`, зберігши підкаталог
`wasm/`. Стенд перевіряє їхні відбитки тим самим браузерним `integrity`, що й продукт.

```bash
export OUT=$(mktemp -d)
export CHROMIUM_PATH=/usr/bin/chromium
photos="аудит/проби/набір-фото-жінок"
for person in ж1 ж2 ж3 ж4 ж7 ж8; do
  mkdir -p "$OUT/$person"
  node джерела/стенд_повернення.js "$photos/$person-повний-зріст.jpg" "$MP" "$OUT/$person"
done

python3 - <<'PY'
from base64 import b64decode
from itertools import product
from pathlib import Path
import json
import os

out = Path(os.environ["OUT"])
photos = Path("аудит/проби/набір-фото-жінок").resolve()
people = ("ж1", "ж2", "ж3", "ж4", "ж7", "ж8")
record = Path("аудит/проби/Прохід-через-міст-2026-11-21/ч3б_примірка/обміни.jsonl")
with record.open() as f:
    exchange = json.loads(next(f))
request = json.loads(exchange["запит"])["messages"][0]["content"]
response = json.loads(exchange["відповідь"])["content"][0]
for name, block in (("photo.jpg", request[0]), ("tryon.png", response)):
    (out / name).write_bytes(b64decode(block["source"]["data"], validate=True))

def pair(name, photo, frame, same, makeup=False):
    return dict(назва=name, фото=str(photo), кадр=str(frame), та_сама=same, макіяж=makeup)

photo = {person: photos / f"{person}-повний-зріст.jpg" for person in people}
actual = [pair("архів 21.11", "photo.jpg", "tryon.png", True, bool(json.loads(request[-1]["text"]).get("makeup")))]
actual += [pair(f"{person} → архів", photo[person], "tryon.png", False) for person in people]
synthetic = [pair(f"{left} → {right}, {kind}", photo[left], f"{right}/{kind}_примірка.png", left == right)
             for left, right, kind in product(people, people, ("колір", "крій"))]
for name, pairs in (("архів", actual), ("синтетика", synthetic)):
    (out / f"{name}.json").write_text(json.dumps(pairs, ensure_ascii=False))
PY

node джерела/стенд_повернення.js --пари "$OUT/архів.json" "$MP"
node джерела/стенд_повернення.js --пари "$OUT/синтетика.json" "$MP"
```

Останній запуск повертає код 1, бо є хибні прийняття і відмови. Файли архівної пари залишаються в тимчасовій теці.
