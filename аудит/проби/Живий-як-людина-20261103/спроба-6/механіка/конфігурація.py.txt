from pathlib import Path
import os,re,json
s=Path(os.environ['TEMP'],'index.html').read_text(encoding='utf-8')
pattern=r'''([\wА-Яа-яІіЇїЄєҐґ_]+)\s*[:=]\s*(["'])gemini-3\.8-flash\2'''
pos=0; names=[]
while True:
 pos=s.find('gemini-3.8-flash',pos)
 if pos<0:break
 names.extend(m.group(1) for m in re.finditer(pattern,s[max(0,pos-100):pos+20]))
 pos+=16
print(json.dumps({'налаштування':names,'входжень_назви':s.count('gemini-3.8-flash')},ensure_ascii=False))
props=set(re.findall(r'window\.([\wА-Яа-яІіЇїЄєҐґ_]+)',s))
print(json.dumps({'глобальні_налаштування':[x for x in sorted(props) if re.search('model|mova|стенд|модел|мова|config|конфіг',x,re.I)]},ensure_ascii=False))
keys=set(re.findall(r'''\.(?:get|getItem|has)\(["']([^"'\n]{1,100})["']\)''',s))
print(json.dumps({'ключі_налаштувань':[x for x in sorted(keys) if re.search('model|mova|стенд|модел|мова|міст|bridge|url',x,re.I)]},ensure_ascii=False))
items=[]
for m in re.finditer(r'''\.(get|getItem|has)\(["']([^"'\n]{1,100})["']\)''',s):
 if m.group(2) in ('model_mova','модель_мови','міст:адреса','міст:модель','міст:модель_мови'):
  prev=re.search(r'([\wА-Яа-яІіЇїЄєҐґ_]+)$',s[max(0,m.start()-100):m.start()])
  items.append({'сховище':prev.group(1) if prev else None,'метод':m.group(1),'ключ':m.group(2)})
print(json.dumps(items,ensure_ascii=False))
