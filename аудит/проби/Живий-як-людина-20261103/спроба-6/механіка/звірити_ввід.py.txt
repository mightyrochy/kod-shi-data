from pathlib import Path
import json,hashlib,sys
base=Path('аудит/проби/Живий-як-людина-20261103/спроба-6')
case=base/sys.argv[1]
def read(run):
 texts=[];photos=[];fields=[]
 for p in sorted((case/run/'скріни').glob('*.json')):
  d=json.loads(p.read_text(encoding='utf-8'));a=json.loads(d['action'])
  if a.get('op')=='fill' and a.get('selector')=='#чат-поле':texts.append(a['value'])
  if a.get('op') in ('fill','select') and a.get('selector')!='#чат-поле':fields.append({'дія':a['op'],'поле':a['selector'],'значення':a['value']})
  if a.get('op')=='file':
   f=Path(a['file']);photos.append({'вхід':a['selector'],'файл':f.name,'sha256':hashlib.sha256(f.read_bytes()).hexdigest()})
 return {'власні_введені_репліки':texts,'завантажені_фото':photos,'введені_поля_форм':fields}
first=read('запит-1');second=read('запит-2')
out={'предмет':'Звірка власного вводу; не нативний запис продукту і не оцінка його відповідей','ситуація':sys.argv[1],'репліки_тотожні':first['власні_введені_репліки']==second['власні_введені_репліки'],'фото_тотожні':first['завантажені_фото']==second['завантажені_фото'],'поля_форм_тотожні':first['введені_поля_форм']==second['введені_поля_форм'],'перший':first,'другий':second}
common=0
for left,right in zip(first['власні_введені_репліки'],second['власні_введені_репліки']):
 if left!=right:break
 common+=1
out['однаковий_початок_реплік']=common
out['додаткові_репліки_першого']=first['власні_введені_репліки'][common:]
out['додаткові_репліки_другого']=second['власні_введені_репліки'][common:]
folder=base/'звірка-вводу';folder.mkdir(exist_ok=True);target=folder/(sys.argv[1]+'.json');target.write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'файл':str(target),'реплік':[len(first['власні_введені_репліки']),len(second['власні_введені_репліки'])],'репліки_тотожні':out['репліки_тотожні'],'фото_тотожні':out['фото_тотожні'],'поля_форм_тотожні':out['поля_форм_тотожні']},ensure_ascii=False))
