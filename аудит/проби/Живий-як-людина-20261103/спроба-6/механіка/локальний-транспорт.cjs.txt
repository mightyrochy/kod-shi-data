const fs=require('fs'),path=require('path');
module.exports={
init:()=>{
 localStorage.setItem('міст:модель','qwen/qwen3.5-9b');
 localStorage.setItem('міст:модель_мови','mamaylm-gemma-3-12b-it-v2.0');
 localStorage.setItem('міст:адреса','http://127.0.0.1:41235/bridge');
 const patch=()=>{
  const cache=u=>{try{const a=new URL(typeof u==='string'?u:u.url);return a.hostname==='cdn.jsdelivr.net'&&a.pathname.includes('/pyodide/')?'http://127.0.0.1:41235/pyodide/'+a.pathname.split('/').pop():u;}catch{return u;}};
  const remote=u=>{try{return /workers\.dev|api\.anthropic\.com|generativelanguage\.googleapis\.com/.test(new URL(typeof u==='string'?u:u.url).hostname)}catch{return false;}};
  const f=globalThis.fetch.bind(globalThis);
  globalThis.fetch=(u,o)=>{const dest=remote(u)?'http://127.0.0.1:41235/bridge':cache(u);return f(dest!==u&&u instanceof Request?new Request(dest,u):dest,o);};
  if(globalThis.importScripts){const imp=globalThis.importScripts.bind(globalThis);globalThis.importScripts=(...u)=>imp(...u.map(cache));}
 };
 patch();
 const OriginalWorker=window.Worker;
 window.Worker=class extends OriginalWorker{constructor(u,o){const script='('+patch.toString()+')();importScripts('+JSON.stringify(String(u))+');';super(URL.createObjectURL(new Blob([script],{type:'application/javascript'})),o);}};
},
async bridge(req,res,base){
 let raw='';for await(const c of req)raw+=c;
 try{
  const b=JSON.parse(raw),meta=[];
  for(const m of b.messages||[])for(const c of Array.isArray(m.content)?m.content:[{type:'text',text:m.content}])if(c.type==='text'){try{const v=JSON.parse(c.text);if(v.task)meta.push({answer:v.task.answer,keys:Object.keys(v.task),schemaKeys:v.task.answer_schema?Object.keys(v.task.answer_schema):[]});}catch{}}
  const answer=meta.map(m=>m.answer||'').join(' ');
  const model=b.model;
  if(!['qwen/qwen3.5-9b','mamaylm-gemma-3-12b-it-v2.0'].includes(model))throw Error('Не налаштовано локальну модель: '+model);
  fs.appendFileSync(path.join(base,'транспорт-форма.jsonl'),JSON.stringify({time:new Date().toISOString(),keys:Object.keys(b),originalModel:b.model,model,tasks:meta})+'\n');
  const messages=[];if(b.system)messages.push({role:'system',content:typeof b.system==='string'?b.system:b.system.map(x=>x.text||'').join('\n')});
  for(const m of b.messages||[])messages.push({role:m.role,content:typeof m.content==='string'?m.content:m.content.map(c=>c.type==='image'?{type:'image_url',image_url:{url:'data:'+c.source.media_type+';base64,'+c.source.data}}:c)});
  const response=await fetch('http://127.0.0.1:41237/v1/chat/completions',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({model,messages,max_tokens:b.max_tokens||4096,temperature:b.temperature??0.3,stream:false})});
  const v=await response.json();if(!response.ok)throw Error('LM Studio '+response.status+': '+JSON.stringify(v));
  const text=v.choices?.[0]?.message?.content||'';
  fs.appendFileSync(path.join(base,'транспорт-кінець.jsonl'),JSON.stringify({time:new Date().toISOString(),answer,model,finish:v.choices?.[0]?.finish_reason,hasContent:!!text})+'\n');
  res.writeHead(200,{'Content-Type':'application/json','Access-Control-Allow-Origin':'*'}).end(JSON.stringify({id:v.id,type:'message',role:'assistant',content:[{type:'text',text}],model,stop_reason:v.choices?.[0]?.finish_reason==='length'?'max_tokens':'end_turn',usage:{input_tokens:v.usage?.prompt_tokens||0,output_tokens:v.usage?.completion_tokens||0}}));
 }catch(e){fs.appendFileSync(path.join(base,'транспорт-помилки.log'),new Date().toISOString()+' '+e.message+'\n');res.writeHead(503,{'Content-Type':'application/json','Access-Control-Allow-Origin':'*'}).end(JSON.stringify({error:{message:e.message}}));}
}
};
