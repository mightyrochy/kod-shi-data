import sys, json, time, uuid
from pathlib import Path
p=Path(__file__).parent/'browser-ipc'; p.mkdir(exist_ok=True)
args=sys.argv[1:]; only_controls='--controls' in args; brief='--brief' in args
args=[a for a in args if a not in ('--controls','--brief')]; action=args[0]
cmd={'id':str(time.time_ns()),'action':action}
if action=='open': cmd['folder']=args[1]
elif action=='upload': cmd['selector']=args[1]; cmd['path']=str(Path(args[2]).resolve())
elif action=='goto': cmd['url']=args[1]
elif action in ('click','fill','select','press'):
    if args[1].startswith('text~'): cmd['text']=args[1][5:]; cmd['exact']=False
    elif args[1].startswith('text='): cmd['text']=args[1][5:]
    else: cmd['selector']=args[1]
    if action in ('fill','select'): cmd['value']=Path(args[2][1:]).read_text(encoding='utf-8') if args[2].startswith('@') else args[2]
    if action=='press': cmd['key']=args[2]
    if action=='click' and len(args)>2: cmd['nth']=int(args[2])
elif action=='scroll': cmd['dy']=int(args[1]) if len(args)>1 else 650
elif action=='wait': cmd['ms']=int(args[1]) if len(args)>1 else 1000
elif action=='note': cmd['text']=args[1]
f=p/(cmd['id']+'.cmd.json'); f.write_text(json.dumps(cmd,ensure_ascii=False),encoding='utf-8')
result=p/(cmd['id']+'.result.json')
for _ in range(450):
    if result.exists():
        try: d=json.loads(result.read_text(encoding='utf-8'))
        except json.JSONDecodeError:
            time.sleep(.1); continue
        print(d.get('time_utc',''), d.get('error',''))
        body=d.get('text','')
        if not only_controls: print(body[:700] if brief else body if len(body)<4000 else body[:500]+'\n… Повний текст у хід.md …\n'+body[-2500:])
        for c in d.get('controls',[]):
            if not brief and (c.get('id') or c.get('text')): print(json.dumps({k:c.get(k) for k in ['tag','id','text','placeholder']},ensure_ascii=False))
        break
    time.sleep(.1)
else: print('Команда ще виконується: '+cmd['id'])
